Build Your Own ISO

# Requirements
sudo apt-get install -y xorriso isolinux

# Build from Ubuntu ISO
UBUNTU_ISO=ubuntu-24.04.2-live-server-amd64.iso \
./live/build-live.sh

# Optional: bake repo into ISO for offline install
REPO_TARBALL=../dist/freeai-v1.2.0.tar.gz \
./live/build-live.sh
